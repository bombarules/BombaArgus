name: Build Bomba APK

on:
  push:
    branches: [ main ]
  workflow_dispatch:

permissions:
  contents: read
  pages: write
  id-token: write

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Unzip project
        run: unzip -o Bomba-github.zip

      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'

      - uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: '8.7'

      - name: Build APK
        working-directory: Bomba
        run: gradle assembleRelease --no-daemon

      - uses: actions/upload-artifact@v4
        with:
          name: Bomba-apk
          path: Bomba/app/build/outputs/apk/release/*.apk

      - name: Prepare Bomba Live web page
        uses: actions/upload-pages-artifact@v3
        with:
          path: Bomba/web

  live-page:
    needs: build
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deploy.outputs.page_url }}
    steps:
      - id: deploy
        uses: actions/deploy-pages@v4
